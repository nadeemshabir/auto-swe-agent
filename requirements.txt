# resolve different dependency versions — a reproducibility problem that becomes
# a deployment problem the moment images are built in CI rather than on your

# work on, captured 2026-07-30 on Python 3.12 (matching the 3.12-slim-bookworm
# base used by all three Dockerfiles).

# To upgrade deliberately:
#     pip install -U <pkg> && pip show <pkg>   # then update the pin below
#     python -m pytest tests/                  # and re-run the suites
#     for m in agent.schemas agent.planner agent.reviewer agent.loop \
#              agent.github db.models workers.tasks app.main; do python -m $m; done
